# CLASS LAB — Window Functions

### Window Function — ROW_NUMBER, RANK and DENSE_RANK

```sql
SELECT ID, NAME, SALARY,
       ROW_NUMBER() OVER (ORDER BY SALARY DESC) AS Row_Num,
       RANK() OVER (ORDER BY SALARY DESC) AS Rnk,
       DENSE_RANK() OVER (ORDER BY SALARY DESC) AS Dense_Rnk
FROM Employees1;
```

### Window Function — DENSE_RANK by Department

```sql
SELECT ID, NAME, DEPARTMENT, SALARY,
       DENSE_RANK() OVER (
           PARTITION BY DEPARTMENT
           ORDER BY SALARY DESC
       ) AS Dept_Rank
FROM Employees1;
```

### Window Function — LAG previous salary

```sql
SELECT NAME, SALARY,
       LAG(SALARY, 1) OVER (
           ORDER BY SALARY DESC
       ) AS Previous_Salary
FROM Employees1;
```

# PL/SQL — Basic Block

### Display employee name and salary using %TYPE

```sql
SET SERVEROUTPUT ON;

DECLARE
    v_emp_id NUMBER := 101;
    v_emp_name Employees1.NAME%TYPE;
    v_salary NUMBER(10,2);
BEGIN
    SELECT NAME, SALARY
    INTO v_emp_name, v_salary
    FROM Employees1
    WHERE ID = v_emp_id;

    DBMS_OUTPUT.PUT_LINE(
        'Employee Name: ' || v_emp_name ||
        ' | Salary: ' || v_salary
    );
END;
/
```

### Implicit Cursor — Increase IT employees' salary by 10% and check SQL%FOUND / SQL%ROWCOUNT

```sql
SET SERVEROUTPUT ON;

DECLARE
    v_rows_updated NUMBER;
BEGIN
    UPDATE Employees1
    SET Salary = Salary + (Salary * 0.10)
    WHERE Department = 'IT';

    IF SQL%FOUND THEN
        v_rows_updated := SQL%ROWCOUNT;

        DBMS_OUTPUT.PUT_LINE(
            'Total ' || v_rows_updated ||
            ' employees salary updated.'
        );
    ELSE
        DBMS_OUTPUT.PUT_LINE(
            'No employee information updated.'
        );
    END IF;

    COMMIT;
END;
/
```

# Explicit Cursor

### Explicit Cursor — Display IT employees

```sql
SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_emp IS
        SELECT ID, NAME, SALARY
        FROM Employees1
        WHERE DEPARTMENT = 'IT';

    v_id Employees1.ID%TYPE;
    v_name Employees1.NAME%TYPE;
    v_salary Employees1.SALARY%TYPE;

BEGIN
    OPEN c_emp;

    LOOP
        FETCH c_emp INTO v_id, v_name, v_salary;

        EXIT WHEN c_emp%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE(
            'ID: ' || v_id ||
            ' | Name: ' || v_name ||
            ' | Salary: ' || v_salary
        );
    END LOOP;

    CLOSE c_emp;
END;
/
```

### Cursor FOR LOOP — Display all employees

```sql
SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_emp IS
        SELECT ID, NAME, SALARY
        FROM Employees1;

BEGIN
    FOR emp_rec IN c_emp LOOP
        DBMS_OUTPUT.PUT_LINE(
            'Name: ' || emp_rec.NAME ||
            ' - Salary: ' || emp_rec.SALARY
        );
    END LOOP;
END;
/
```

# TRIGGERS

### Trigger 1 — AFTER INSERT audit log

```sql
CREATE TABLE student_audit_table (
    student_id NUMBER,
    name VARCHAR2(50),
    age NUMBER
);

CREATE TABLE student_log (
    student_id NUMBER,
    name VARCHAR2(50),
    log_date DATE
);

CREATE OR REPLACE TRIGGER trg_student_insert
AFTER INSERT ON student_audit_table
FOR EACH ROW
BEGIN
    INSERT INTO student_log
    VALUES (:NEW.student_id, :NEW.name, SYSDATE);
END;
/
```

### Test Trigger 1

```sql
INSERT INTO student_audit_table
VALUES (102, 'Rahim', 22);

SELECT * FROM student_log;
```

### Trigger 2 — AFTER UPDATE salary audit

```sql
CREATE TABLE employee_pay (
    emp_id NUMBER,
    name VARCHAR2(50),
    salary NUMBER
);

CREATE TABLE salary_log (
    emp_id NUMBER,
    old_salary NUMBER,
    new_salary NUMBER,
    change_date DATE
);

INSERT INTO employee_pay
VALUES (101, 'Rahim', 30000);

COMMIT;

CREATE OR REPLACE TRIGGER trg_salary_update
AFTER UPDATE OF salary ON employee_pay
FOR EACH ROW
BEGIN
    INSERT INTO salary_log
    VALUES (
        :OLD.emp_id,
        :OLD.salary,
        :NEW.salary,
        SYSDATE
    );
END;
/
```

### Test Trigger 2

```sql

SET salary = 50000
WHERE emp_id = 101;

SELECT * FROM salary_log;
```

### Trigger 3 — BEFORE INSERT automatic salary correction

### Trigger 4 — BEFORE UPDATE negative salary prevention


### Trigger 5 — BEFORE DELETE notification and audit log

```sql
CREATE OR REPLACE TRIGGER trg_before_delete
BEFORE DELETE ON employee_pay
FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE(
        'Employee ' || :OLD.name ||
        ' is about to be DELETED!'
    );

    INSERT INTO salary_log
    VALUES (
        :OLD.emp_id,
        :OLD.salary,
        NULL,
        SYSDATE
    );
END;
/
```
